4. Teacher Oracles
Manhattan Snake Oracle
Purpose: Greedy pathfinding toward food with obstacle avoidance.

def manhattan_snake_move(state: dict, obstacles: set) -> int
Parameters:

state (dict): {"head": (x, y), "food": (fx, fy)}
obstacles (set): Set of (x, y) coordinates to avoid
Algorithm:

head_x, head_y = state["head"]
food_x, food_y = state["food"]

best_action = None
min_distance = infinity

for action_idx, (dx, dy) in enumerate(DIRECTIONS):
   next_x = (head_x + dx) % GRID_SIZE
   next_y = (head_y + dy) % GRID_SIZE
   
   if (next_x, next_y) in obstacles:
       continue  # Skip blocked moves
   
   distance = |next_x - food_x| + |next_y - food_y|
   
   if distance < min_distance:
       min_distance = distance
       best_action = action_idx

return best_action
Complexity: O(4) = O(1) for 4 directions

def get_pong_oracle(state: dict) -> int

5. Main Loop
Purpose: Core event loop processing game inputs.

def main():
   # Setup ZMQ
   pull_sock = zmq.PULL on 5555
   pub_sock = zmq.PUB on 5556
   pub_stats = zmq.PUB on 5557
   
   # Initialize
   model = TaskAwareSNN()
   optimizer = Adam(lr=LEARNING_RATE)
   buffer = ReplayBuffer()
   logger = LogAggregator()
   
   while True:
       msg = pull_sock.recv_json()
       
       # Admin command handling
       if msg["type"] == "admin":
           handle_admin(msg["cmd"])
           continue
       
       # Extract game state
       game_type = msg["game"]
       state = msg["state"]
       image = decode_base64_png(msg["image"])
       
       # Task switch detection
       if game_type != last_game_type:
           sleep_cycle(model, optimizer, buffer, device)
       
       # Process input
       frames = build_temporal_stack(image, history[game_type])
       input_tensor = torch.from_numpy(frames).unsqueeze(0)
       
       # Teacher prediction
       teacher_action = oracle(game_type, state, image)
       
       # Student inference
       task_id = 1 if game_type == "snake" else 0
       output = model(input_tensor, task_id)
       student_action = argmax(softmax(output))
       
       # Disagreement gate
       if student_action == teacher_action:
           # Agreement: No training
           final_action = student_action
           loss = 0.0
       else:
           # Disagreement: Train and intervene
           final_action = teacher_action
           loss = cross_entropy(output, teacher_action)
           loss.backward()
           optimizer.step()
       
       # Store experience
       buffer.push(input_tensor, task_id, teacher_action, 
                  agreed=(student_action == teacher_action), game_type)
       
       # Update statistics
       logger.update(game_type, agreed, loss)
       logger.check_print()
       
       # Publish action
       pub_sock.send_string(f"{game_type.upper()}:{ACTION_NAMES[final_action]}")
       
       # Periodic telemetry
       if steps % 10 == 0:
           send_visualization(pub_stats, image, output, teacher_action, 
                            student_action, agreed)
       
       # Save checkpoint
       if steps % SAVE_INTERVAL == 0:
           torch.save(model.state_dict(), MODEL_PATH)

Forward Pass
def forward(self, x: torch.Tensor, task_id: int) -> torch.Tensor
Parameters:

x (torch.Tensor): Input tensor [Batch, 4, 10, 10]
task_id (int): 0 (Pong) or 1 (Snake)
Returns:

torch.Tensor: Output logits [Batch, num_actions]
[Batch, 4] for Snake
[Batch, 2] for Pong
Algorithm:

# Initialize membrane potentials
mem1 = self.lif1.init_leaky()
mem2 = self.lif2.init_leaky()
mem_shared = self.lif_shared.init_leaky()
mem_out = self.lif_out.init_leaky()

spike_recordings = []

# Simulate for T timesteps
for t in range(T=8):
   # Visual cortex
   cur1 = self.conv1(x)
   spk1, mem1 = self.lif1(cur1, mem1)
   
   cur2 = self.conv2(spk1)
   spk2, mem2 = self.lif2(cur2, mem2)
   
   # Late fusion
   flat = self.flatten(spk2)
   task_vec = torch.full((x.size(0), 1), float(task_id), device=x.device)
   fused = torch.cat([flat, task_vec], dim=1)
   
   cur_shared = self.fc_shared(fused)
   spk_shared, mem_shared = self.lif_shared(cur_shared, mem_shared)
   
   # Task head
   if task_id == 1:
       cur_out = self.head_snake(spk_shared)
   else:
       cur_out = self.head_pong(spk_shared)
   
   spk_out, mem_out = self.lif_out(cur_out, mem_out)
   spike_recordings.append(spk_out)

# Rate coding: Sum spikes over time
output = torch.stack(spike_recordings, dim=0).sum(0)
return output

LIF Neuron Dynamics
Implemented by: snntorch.Leaky

Discrete-Time Update:

# Current injection
I[t] = W @ x[t]

# Membrane potential update
U[t] = beta * U[t-1] + I[t]

# Spike generation
if U[t] >= threshold:
   S[t] = 1
   U[t] = U[t] - threshold  # Soft reset
else:
   S[t] = 0

return S[t], U[t]
Surrogate Gradient (Backward):

# Forward: Heaviside step function
grad_forward = H(U - theta)

# Backward: Fast sigmoid approximation
grad_backward = k * sigma(k*U) * (1 - sigma(k*U))

where sigma(x) = 1 / (1 + exp(-x))
     k = slope parameter (default 25)
Python Implementation:

class FastSigmoid(torch.autograd.Function):
   @staticmethod
   def forward(ctx, input, slope=25):
       ctx.save_for_backward(input)
       ctx.slope = slope
       return (input > 0).float()
   
   @staticmethod
   def backward(ctx, grad_output):
       input, = ctx.saved_tensors
       slope = ctx.slope
       grad_input = grad_output / (slope * torch.abs(input) + 1.0) ** 2
       return grad_input, None

__init__(seed)
Generates random hypervector.

def __init__(self, seed=None):
   if seed is not None:
       self.rng = np.random.default_rng(seed)
   else:
       self.rng = np.random.default_rng()
   
   self.bits = self.rng.integers(0, 2, size=DIMENSION, dtype=np.int8)
xor(other: HyperVector) -> HyperVector
Binding operation.

def xor(self, other):
   new_bits = np.bitwise_xor(self.bits, other.bits)
   return HyperVector.from_bits(new_bits)
Example:

apple = HyperVector(seed=1)
north = HyperVector(seed=2)
target = apple.xor(north)  # Encode "APPLE is NORTH"

# Decode: target XOR apple ≈ north

bundle(other: HyperVector) -> HyperVector
Superposition operation with random tie-breaking.

def bundle(self, other):
   rand_mask = np.random.randint(0, 2, size=DIMENSION, dtype=np.int8)
   
   a = self.bits
   b = other.bits
   diff = np.bitwise_xor(a, b)
   same = np.bitwise_and(a, b)
   
   # Where bits agree: keep. Where differ: random choice
   bundle_bits = np.bitwise_or(same, np.bitwise_and(diff, rand_mask))
   return HyperVector.from_bits(bundle_bits)
Example:

apple = HyperVector(seed=1)
banana = HyperVector(seed=2)
fruit = apple.bundle(banana)  # Superposition

print(fruit.similarity(apple))   # ~0.75
print(fruit.similarity(banana))  # ~0.75
similarity(other: HyperVector) -> float
Normalized Hamming similarity.

def similarity(self, other):
   diff = np.bitwise_xor(self.bits, other.bits)
   hamming_dist = np.sum(diff)
   return 1.0 - (hamming_dist / DIMENSION)
Range: [0.0, 1.0]

1.0 = Identical
0.5 = Random/Orthogonal
0.0 = Perfectly anti-correlated (rare)

HyperVector Struct
#[pyclass]
#[derive(Clone, Debug)]
struct HyperVector {
   bits: Vec<u64>,  // 160 u64 blocks for 10,240 bits
}
Storage: 160 × 8 bytes = 1,280 bytes per vector

Methods
new(seed: Option<u64>) -> Self
#[new]
fn new(seed: Option<u64>) -> Self {
   let mut rng = match seed {
       Some(s) => ChaCha8Rng::seed_from_u64(s),
       None => ChaCha8Rng::from_entropy(),
   };

   let num_u64 = DIMENSION / 64;  // 160
   let mut bits = Vec::with_capacity(num_u64);
   
   for _ in 0..num_u64 {
       bits.push(rng.gen());  // Random u64
   }

   HyperVector { bits }
}
xor(&self, other: &HyperVector) -> HyperVector
fn xor(&self, other: &HyperVector) -> HyperVector {
   let fused: Vec<u64> = self.bits.iter()
       .zip(other.bits.iter())
       .map(|(a, b)| a ^ b)  // XOR each u64 block
       .collect();
   HyperVector { bits: fused }
}
Performance: ~1 microsecond

similarity(&self, other: &HyperVector) -> f64
fn similarity(&self, other: &HyperVector) -> f64 {
   let mut hamming_dist: u32 = 0;
   
   for (a, b) in self.bits.iter().zip(other.bits.iter()) {
       hamming_dist += (a ^ b).count_ones();  // Population count
   }
   
   1.0 - (hamming_dist as f64 / DIMENSION as f64)
}
Performance: ~10 microseconds (with count_ones hardware instruction)

step(observation_idx: int, precision_signal: float = 1.0) -> int
Performs one inference-action cycle.

def step(self, observation_idx, precision_signal=1.0):
   # State inference
   obs = [observation_idx]
   qs = self.agent.infer_states(obs)  # Posterior belief over states
   
   # Policy selection (minimize expected free energy)
   q_pi, neg_efe = self.agent.infer_policies()
   
   # Sample action
   action = self.agent.sample_action()  # Sample from policy distribution
   
   return int(action[0])
Expected Free Energy (EFE):

G(π) = E_Q(o,s|π)[log Q(s|π) - log P(o, s)]
    = -Epistemic_Value - Pragmatic_Value

Epistemic Value: Information gain (exploration)
Pragmatic Value: Expected reward from C-matrix (exploitation)
Policy Distribution:

P(π) = σ(-γ * G(π))

where σ = softmax, γ = precision (inverse temperature)
